An efficient, self-contained Python script to compute the total sum of pairwise Euclidean distances is provided below:

```python
import numpy as np


def main():
    # Load the vectors matrix (8000 x 1024 float32)
    X = np.load("vectors.npy").astype(np.float64)

    # Compute squared norms of each vector: shape (N, 1)
    sq_norms = np.sum(X**2, axis=1, keepdims=True)

    # Compute total pairwise Euclidean distance using blocked matrix multiplication
    # to maintain high precision and low memory overhead.
    total_dist = 0.0
    batch_size = 2000
    n = X.shape[0]

    for i in range(0, n, batch_size):
        X_batch = X[i : i + batch_size]
        sq_batch = sq_norms[i : i + batch_size]

        # Pairwise squared distances: ||u - v||^2 = ||u||^2 + ||v||^2 - 2<u, v>
        # X_batch @ X.T computes all inner products for the current batch
        dists_sq = sq_batch + sq_norms.T - 2.0 * np.dot(X_batch, X.T)

        # Numerical safety: clip negative values resulting from precision issues
        np.maximum(dists_sq, 0.0, out=dists_sq)

        # Compute Euclidean distance and accumulate
        total_dist += np.sum(np.sqrt(dists_sq))

    print(f"TOTAL_DIST:{total_dist}")


if __name__ == "__main__":
    main()
```
