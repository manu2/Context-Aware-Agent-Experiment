An efficient and self-contained Python script to compute the total pairwise Euclidean distance sum is provided below:

```python
import numpy as np


def main():
    # Load vectors (8000 x 1024, float32) and cast to float64 for numerical precision
    vectors = np.load("vectors.npy").astype(np.float64)
    n = vectors.shape[0]

    # Precompute squared L2 norms for each vector
    norms_sq = np.sum(vectors**2, axis=1)

    total_dist = 0.0
    chunk_size = 1000

    # Process in chunks to maintain low memory footprint and high BLAS performance
    for i in range(0, n, chunk_size):
        end_i = min(i + chunk_size, n)
        v_chunk = vectors[i:end_i]
        norms_chunk = norms_sq[i:end_i, np.newaxis]

        # ||v_i - v_j||^2 = ||v_i||^2 + ||v_j||^2 - 2 * (v_i . v_j)
        dists_sq = (
            norms_chunk + norms_sq[np.newaxis, :] - 2.0 * np.dot(v_chunk, vectors.T)
        )
        np.maximum(dists_sq, 0.0, out=dists_sq)

        # Explicitly zero out self-distances on the diagonal to avoid float roundoff
        for local_i in range(end_i - i):
            dists_sq[local_i, i + local_i] = 0.0

        np.sqrt(dists_sq, out=dists_sq)
        total_dist += np.sum(dists_sq)

    print(f"TOTAL_DIST:{total_dist}")


if __name__ == "__main__":
    main()
```
