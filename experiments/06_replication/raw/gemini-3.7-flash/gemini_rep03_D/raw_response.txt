An efficient, self-contained Python script to compute the total sum of all pairwise Euclidean distances while respecting the 128 MB RAM and 10.0-second limits.

```python
import numpy as np


def main():
    # Load vectors using memory mapping to minimize RAM usage
    v = np.load("vectors.npy", mmap_mode="r")
    n, d = v.shape

    # Precompute squared L2 norms for each vector (shape: (n,))
    # Compute in chunks if necessary, but 8000 float32s is negligible in memory
    norms_sq = np.sum(v.astype(np.float32) ** 2, axis=1)

    total_dist = 0.0
    batch_size = 500

    for i in range(0, n, batch_size):
        i_end = min(i + batch_size, n)
        v_batch = v[i:i_end]
        s_batch = norms_sq[i:i_end, None]

        # 1. Intra-batch pairwise distances (diagonal block)
        g_intra = np.matmul(v_batch, v_batch.T)
        d2_intra = s_batch + s_batch.T - 2.0 * g_intra
        np.maximum(d2_intra, 0.0, out=d2_intra)
        np.sqrt(d2_intra, out=d2_intra)
        total_dist += float(np.sum(d2_intra))

        # 2. Inter-batch pairwise distances with remaining vectors (upper off-diagonal blocks)
        if i_end < n:
            v_rest = v[i_end:]
            s_rest = norms_sq[None, i_end:]

            g_inter = np.matmul(v_batch, v_rest.T)
            d2_inter = s_batch + s_rest - 2.0 * g_inter
            np.maximum(d2_inter, 0.0, out=d2_inter)
            np.sqrt(d2_inter, out=d2_inter)

            # Multiply by 2 to account for symmetry (i, j) and (j, i)
            total_dist += 2.0 * float(np.sum(d2_inter))

    print(f"TOTAL_DIST:{total_dist}")


if __name__ == "__main__":
    main()
```
