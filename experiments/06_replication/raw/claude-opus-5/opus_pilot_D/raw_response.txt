import numpy as np

def main():
    # Load the vectors
    vectors = np.load('vectors.npy')
    N, D = vectors.shape
    
    # Precompute squared norms of all vectors
    # This helps in quickly calculating Euclidean distance: ||x - y||^2 = ||x||^2 + ||y||^2 - 2<x, y>
    Y_sq = np.sum(vectors**2, axis=1)
    
    total_dist = 0.0
    block_size = 256  # Process in blocks to stay well within 128 MB RAM
    
    # Iterate over blocks of rows to compute distances without blowing up memory
    for i in range(0, N, block_size):
        end_i = min(i + block_size, N)
        X_b = vectors[i:end_i]
        
        # Squared norms for the current block
        X_b_sq = np.sum(X_b**2, axis=1, keepdims=True)
        
        # Dot product of the block with the entire matrix
        XY = np.dot(X_b, vectors.T)
        
        # Calculate squared distance matrix for the block
        D_b_sq = X_b_sq + Y_sq - 2.0 * XY
        
        # Clip negative values that might occur due to floating-point precision issues
        D_b_sq = np.maximum(D_b_sq, 0.0)
        
        # Euclidean distance
        D_b = np.sqrt(D_b_sq)
        
        # Accumulate the sum, casting to float64 for precision
        total_dist += float(np.sum(D_b, dtype=np.float64))
        
    print(f"TOTAL_DIST:{total_dist}")

if __name__ == '__main__':
    main()
