An optimal and exact solution in Python using NumPy:

```python
import numpy as np


def main():
    # Load vectors and convert to float64 to ensure numerical accuracy
    vectors = np.load("vectors.npy").astype(np.float64)

    # Compute squared Euclidean norms of each vector
    norms = np.sum(vectors**2, axis=1)

    # Compute pairwise squared distances: ||u - v||^2 = ||u||^2 + ||v||^2 - 2*(u . v)
    # Reusing the Gram matrix array in-place keeps memory footprint minimal (~500 MB)
    dists_sq = vectors @ vectors.T
    dists_sq *= -2.0
    dists_sq += norms[:, None]
    dists_sq += norms[None, :]

    # Clamp any slight numerical inaccuracies to 0 and explicitly zero the diagonal
    np.maximum(dists_sq, 0.0, out=dists_sq)
    np.fill_diagonal(dists_sq, 0.0)

    # Compute Euclidean distances and sum over all pairs
    np.sqrt(dists_sq, out=dists_sq)
    total = float(dists_sq.sum())

    print(f"TOTAL:{total}")


if __name__ == "__main__":
    main()
```
