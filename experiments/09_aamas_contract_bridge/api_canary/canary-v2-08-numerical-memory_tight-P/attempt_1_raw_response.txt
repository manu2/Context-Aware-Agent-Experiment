import numpy as np


def main():
    X = np.load("vectors.npy")
    N = X.shape[0]

    # Precompute squared L2 norm of each row
    norms_sq = np.sum(X * X, axis=1, dtype=np.float64)

    total_dist = 0.0
    block_size = 2000

    for i in range(0, N, block_size):
        i_end = min(i + block_size, N)
        X_i = X[i:i_end]
        norms_i = norms_sq[i:i_end, None]

        for j in range(i, N, block_size):
            j_end = min(j + block_size, N)
            X_j = X[j:j_end]
            norms_j = norms_sq[None, j:j_end]

            # Compute pairwise squared distances: ||u - v||^2 = ||u||^2 + ||v||^2 - 2<u, v>
            gram = X_i @ X_j.T
            dist_sq = norms_i + norms_j - 2.0 * gram
            np.maximum(dist_sq, 0.0, out=dist_sq)
            dist = np.sqrt(dist_sq, out=dist_sq)

            if i == j:
                total_dist += np.sum(np.triu(dist, k=1), dtype=np.float64)
            else:
                total_dist += np.sum(dist, dtype=np.float64)

    # Double to account for both (i, j) and (j, i) pairs; diagonal is 0
    total_dist *= 2.0

    print(f"TOTAL:{total_dist}")


if __name__ == "__main__":
    main()
