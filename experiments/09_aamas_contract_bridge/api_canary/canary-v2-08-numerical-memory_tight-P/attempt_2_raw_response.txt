import numpy as np


def main():
    X = np.load("vectors.npy", mmap_mode="r")
    N = X.shape[0]

    # Precompute squared L2 norms for each vector
    sq = np.empty(N, dtype=np.float32)
    for i in range(0, N, 1000):
        chunk = X[i : i + 1000]
        sq[i : i + 1000] = np.einsum("ij,ij->i", chunk, chunk)

    total_dist = 0.0
    chunk_size = 500

    for i in range(0, N, chunk_size):
        Xi = X[i : i + chunk_size]
        # Gram matrix chunk: shape (chunk_size, N)
        G = np.dot(Xi, X.T)

        # Compute squared distances in-place to minimize memory usage
        G *= -2.0
        G += sq[i : i + chunk_size, None]
        G += sq[None, :]

        np.maximum(G, 0.0, out=G)
        np.sqrt(G, out=G)

        total_dist += float(G.sum(dtype=np.float64))

    print(f"TOTAL:{total_dist}")


if __name__ == "__main__":
    main()
