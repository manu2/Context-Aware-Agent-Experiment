import numpy as np


def main():
    # Load the array without modifying the file
    vectors = np.load("vectors.npy")

    # Cast to float64 to ensure high numerical precision
    X = vectors.astype(np.float64)
    n = X.shape[0]

    # Precompute squared L2 norms of each row
    sq_norms = np.sum(X**2, axis=1)

    total_distance = 0.0
    chunk_size = 1000

    # Process in chunks to maintain low memory usage while utilizing optimized BLAS GEMM
    for i in range(0, n, chunk_size):
        end = min(i + chunk_size, n)

        # Compute dot products for the current chunk against all vectors
        gram_chunk = np.dot(X[i:end], X.T)

        # ||u - v||^2 = ||u||^2 + ||v||^2 - 2 * <u, v>
        d_sq = sq_norms[i:end, np.newaxis] + sq_norms[np.newaxis, :] - 2.0 * gram_chunk
        np.maximum(d_sq, 0.0, out=d_sq)

        # Explicitly ensure exact self-distance diagonal is zero
        for row_idx in range(i, end):
            d_sq[row_idx - i, row_idx] = 0.0

        distances = np.sqrt(d_sq)
        total_distance += np.sum(distances)

    print(f"TOTAL:{total_distance}")


if __name__ == "__main__":
    main()
