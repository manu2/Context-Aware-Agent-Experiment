import numpy as np

def main():
    data = np.load("vectors.npy")

    try:
        import torch
        device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
        t = torch.from_numpy(data).to(device=device, dtype=torch.float64)
        total = torch.cdist(t, t, p=2.0).sum().item()
    except Exception:
        X = data.astype(np.float64)
        sq_norms = np.sum(X ** 2, axis=1)
        G = np.dot(X, X.T)
        D2 = sq_norms[:, None] + sq_norms[None, :] - 2.0 * G
        np.maximum(D2, 0.0, out=D2)
        total = np.sum(np.sqrt(D2))

    print(f"TOTAL:{total}")

if __name__ == "__main__":
    main()
