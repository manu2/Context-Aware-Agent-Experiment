import numpy as np

def main():
    vectors = np.load("vectors.npy").astype(np.float64)
    norms_sq = np.sum(vectors ** 2, axis=1)
    
    total_distance = 0.0
    n = vectors.shape[0]
    chunk_size = 2000
    
    for i in range(0, n, chunk_size):
        chunk = vectors[i : i + chunk_size]
        gram = chunk @ vectors.T
        dist_sq = norms_sq[i : i + chunk.shape[0], None] + norms_sq[None, :] - 2.0 * gram
        np.maximum(dist_sq, 0.0, out=dist_sq)
        total_distance += np.sum(np.sqrt(dist_sq))
    
    print(f"TOTAL:{total_distance}")

if __name__ == "__main__":
    main()
