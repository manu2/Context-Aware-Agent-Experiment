import numpy as np

def main():
    # Load data (float32, 8000 x 1024) -- ~31.25 MiB in memory
    X = np.load('vectors.npy')
    N, D = X.shape

    # Precompute squared norms of each row (float64 for accuracy)
    norms = np.einsum('ij,ij->i', X, X, dtype=np.float64)

    total_upper = 0.0  # sum over i < j of distances

    block_size = 500  # tuned to keep memory usage low

    for a in range(0, N, block_size):
        end_i = min(a + block_size, N)
        bi = end_i - a
        Xi = X[a:end_i]          # shape (bi, D)
        Xj = X[a:N]              # shape (bj, D)
        bj = N - a

        # Dot products between rows in this block and all rows from a onward
        dot = Xi @ Xj.T          # (bi, bj), float32

        ni = norms[a:end_i][:, None]     # (bi,1) float64
        nj = norms[a:N][None, :]         # (1,bj) float64

        # Squared distances (float64 for numerical stability)
        sq = ni + nj - 2.0 * dot.astype(np.float64)
        np.maximum(sq, 0.0, out=sq)
        d = np.sqrt(sq)

        # Build mask for strictly upper part: global j > global i
        # local row r -> global i = a + r
        # local col c -> global j = a + c
        rows = np.arange(bi)[:, None]
        cols = np.arange(bj)[None, :]
        mask = cols > rows

        total_upper += d[mask].sum(dtype=np.float64)

        # free memory explicitly
        del dot, sq, d, mask, ni, nj

    total = 2.0 * total_upper

    print(f"TOTAL:{total:.6f}")

if __name__ == "__main__":
    main()
