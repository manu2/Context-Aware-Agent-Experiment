import numpy as np

def main():
    X = np.load('vectors.npy')  # (8000, 1024) float32, read-only usage
    N, D = X.shape

    # Compute squared norms with double precision for numerical accuracy
    norms = np.einsum('ij,ij->i', X, X, dtype=np.float64)

    total = 0.0
    B = 1000  # block size, chosen to balance memory and speed
    nblocks = (N + B - 1) // B

    for bi in range(nblocks):
        i0, i1 = bi * B, min((bi + 1) * B, N)
        Xi = X[i0:i1]
        ni = norms[i0:i1]

        # ---- diagonal block (bi == bj) ----
        G = Xi @ Xi.T  # float32 matmul (fast BLAS)
        D2 = ni[:, None] + ni[None, :] - 2.0 * G.astype(np.float64)
        np.maximum(D2, 0.0, out=D2)  # guard tiny negative values from rounding
        Dm = np.sqrt(D2)
        # only strict upper triangle counted once, then doubled for symmetry
        s = np.triu(Dm, k=1).sum()
        total += 2.0 * s

        # ---- off-diagonal blocks (bj > bi) ----
        for bj in range(bi + 1, nblocks):
            j0, j1 = bj * B, min((bj + 1) * B, N)
            Xj = X[j0:j1]
            nj = norms[j0:j1]

            G2 = Xi @ Xj.T
            D2b = ni[:, None] + nj[None, :] - 2.0 * G2.astype(np.float64)
            np.maximum(D2b, 0.0, out=D2b)
            Db = np.sqrt(D2b)
            total += 2.0 * Db.sum()

    print(f"TOTAL:{total!r}")

if __name__ == "__main__":
    main()
