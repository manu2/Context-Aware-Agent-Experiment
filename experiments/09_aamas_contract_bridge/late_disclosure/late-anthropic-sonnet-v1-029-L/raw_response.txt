```python
import numpy as np

def main():
    X = np.load("vectors.npy", mmap_mode="r")  # (8000, 1024) float32, ~32MB on disk
    n, d = X.shape

    # Compute squared norms once, in float64, using small chunks to bound memory
    sq_norms = np.empty(n, dtype=np.float64)
    norm_chunk = 1000
    for start in range(0, n, norm_chunk):
        end = min(start + norm_chunk, n)
        block = np.asarray(X[start:end], dtype=np.float64)
        sq_norms[start:end] = np.einsum('ij,ij->i', block, block)

    total = 0.0
    row_chunk = 200   # number of rows processed against full dataset at once
    col_chunk = 2000  # number of columns (other rows) processed at once

    for i_start in range(0, n, row_chunk):
        i_end = min(i_start + row_chunk, n)
        A = np.asarray(X[i_start:i_end], dtype=np.float64)  # (r, d)
        A_norms = sq_norms[i_start:i_end]  # (r,)

        row_total = np.zeros(A.shape[0], dtype=np.float64)

        for j_start in range(0, n, col_chunk):
            j_end = min(j_start + col_chunk, n)
            B = np.asarray(X[j_start:j_end], dtype=np.float64)  # (c, d)
            B_norms = sq_norms[j_start:j_end]  # (c,)

            # squared distances: ||a||^2 + ||b||^2 - 2 a.b
            dots = A @ B.T  # (r, c)
            sq_dist = A_norms[:, None] + B_norms[None, :] - 2.0 * dots
            np.maximum(sq_dist, 0.0, out=sq_dist)  # guard tiny negatives from rounding
            dist = np.sqrt(sq_dist)

            row_total += dist.sum(axis=1)

        total += row_total.sum()

    # total currently a numpy float64 scalar; format as decimal
    print(f"TOTAL:{total:.10f}")

if __name__ == "__main__":
    main()
```
