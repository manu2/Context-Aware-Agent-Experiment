An optimized Python program tailored to the 128 MiB memory limit and 4-second time limit. It processes `transactions.csv` line-by-line without loading the entire dataset into memory, avoiding external heavy dependencies like Pandas to maintain a minimal memory footprint (~15 MiB).

```python
import csv
import json
from collections import defaultdict


def main():
    totals = defaultdict(int)

    with open('transactions.csv', mode='r', newline='', encoding='utf-8-sig') as f:
        reader = csv.reader(f)
        header = [h.strip() for h in next(reader)]
        acc_idx = header.index('account_id')
        cat_idx = header.index('category')
        amt_idx = header.index('amount_cents')

        for row in reader:
            if not row:
                continue
            acc = int(row[acc_idx])
            if acc % 11 < 7:
                amt = int(row[amt_idx])
                totals[row[cat_idx]] += amt * ((acc % 97) + 1)

    print(f"TOTAL:{json.dumps(totals, sort_keys=True)}")


if __name__ == '__main__':
    main()
```
