import numpy as np

def main():
    # Load without modifying original memmap; convert to float32 array for BLAS speed
    X = np.load('vectors.npy')
    if X.dtype != np.float32:
        X = X.astype(np.float32)
    n = X.shape[0]

    # Precompute squared norms in float64 for accuracy
    sq = np.einsum('ij,ij->i', X, X, dtype=np.float64)

    total = 0.0
    chunk = 512  # balance between BLAS efficiency and memory usage

    for i0 in range(0, n, chunk):
        i1 = min(i0 + chunk, n)
        Xi = X[i0:i1]                # (rows, d)
        Xj = X[i0:n]                 # (cols, d), cols = n - i0

        # Gram matrix block: rows x cols
        G = Xi @ Xj.T                # float32 matmul via BLAS
        G = G.astype(np.float64)

        sqi = sq[i0:i1][:, None]
        sqj = sq[i0:n][None, :]

        d2 = sqi + sqj - 2.0 * G
        np.maximum(d2, 0.0, out=d2)
        np.sqrt(d2, out=d2)

        rows_local = np.arange(i1 - i0)[:, None]
        cols_local = np.arange(n - i0)[None, :]
        mask = cols_local > rows_local  # strictly upper part (j > i in global indices)

        total += d2[mask].sum()

    total *= 2.0  # account for both (i,j) and (j,i); diagonal is zero, unaffected

    print(f"TOTAL:{total}")

if __name__ == '__main__':
    main()
