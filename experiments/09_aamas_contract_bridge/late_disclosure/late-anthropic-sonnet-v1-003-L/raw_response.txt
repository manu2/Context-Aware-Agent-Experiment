```python
import csv
import json
from collections import defaultdict

totals = defaultdict(int)

with open('transactions.csv', newline='') as f:
    reader = csv.reader(f)
    header = next(reader)
    idx_account = header.index('account_id')
    idx_category = header.index('category')
    idx_amount = header.index('amount_cents')

    for row in reader:
        if not row:
            continue
        account_id = int(row[idx_account])
        if account_id % 11 < 7:
            amount_cents = int(row[idx_amount])
            multiplier = (account_id % 97) + 1
            totals[row[idx_category]] += amount_cents * multiplier

result = {k: totals[k] for k in sorted(totals.keys())}
print("TOTAL:" + json.dumps(result))
```
