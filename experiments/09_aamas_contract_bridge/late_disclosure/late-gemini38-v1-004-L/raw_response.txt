import numpy as np

def main():
    X = np.load("vectors_secondary.npy", mmap_mode="r")
    N = X.shape[0]

    sq_norms = np.empty(N, dtype=np.float64)
    for i in range(0, N, 1000):
        sq_norms[i : i + 1000] = np.sum(
            X[i : i + 1000].astype(np.float64) ** 2, axis=1
        )

    total_half_sum = 0.0
    B = 400

    for i in range(0, N, B):
        i_end = min(i + B, N)
        b = i_end - i

        dot = X[i:i_end] @ X[i:].T
        d2 = sq_norms[i:i_end, None] + sq_norms[None, i:] - 2.0 * dot
        np.maximum(d2, 0.0, out=d2)
        np.sqrt(d2, out=d2)

        total_half_sum += np.sum(np.triu(d2[:, :b], k=1))
        if b < d2.shape[1]:
            total_half_sum += np.sum(d2[:, b:])

    total_sum = 2.0 * total_half_sum
    print(f"TOTAL:{total_sum}")

if __name__ == "__main__":
    main()
