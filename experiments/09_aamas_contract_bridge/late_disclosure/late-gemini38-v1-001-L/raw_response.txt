```python
from collections import defaultdict
import json
import warnings
import pandas as pd

warnings.filterwarnings("ignore")


def main():
    totals = defaultdict(int)

    for chunk in pd.read_csv(
        "transactions_secondary.csv",
        usecols=["account_id", "category", "amount_cents"],
        dtype={"account_id": "int64", "category": "str", "amount_cents": "int64"},
        chunksize=500_000,
    ):
        mask = (chunk["account_id"] % 13) < 8
        filtered = chunk[mask]
        if not filtered.empty:
            mult = (filtered["account_id"] % 89) + 3
            products = filtered["amount_cents"] * mult
            chunk_totals = products.groupby(filtered["category"]).sum()
            for cat, val in chunk_totals.items():
                totals[str(cat)] += int(val)

    result = dict(sorted(totals.items()))
    print(f"TOTAL:{json.dumps(result, sort_keys=True)}")


if __name__ == "__main__":
    main()
```
