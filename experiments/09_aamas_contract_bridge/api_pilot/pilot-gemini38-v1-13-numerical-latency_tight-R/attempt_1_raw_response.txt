import numpy as np


def main():
    X = np.load("vectors.npy").astype(np.float64)

    # Compute squared Euclidean norms of each row
    norms_sq = np.sum(X**2, axis=1)

    # Compute Gram matrix: G = X @ X.T
    G = X @ X.T

    # Compute squared distance matrix in-place: ||u - v||^2 = ||u||^2 + ||v||^2 - 2<u, v>
    G *= -2.0
    G += norms_sq[:, None]
    G += norms_sq[None, :]

    # Eliminate small negative values caused by numerical inaccuracy and zero the diagonal
    np.maximum(G, 0.0, out=G)
    np.fill_diagonal(G, 0.0)

    # Compute Euclidean distances
    np.sqrt(G, out=G)

    total = np.sum(G)
    print(f"TOTAL:{total}")


if __name__ == "__main__":
    main()
