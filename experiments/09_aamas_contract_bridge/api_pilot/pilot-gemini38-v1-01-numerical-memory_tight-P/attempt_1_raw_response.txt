import numpy as np


def main():
    X = np.load("vectors.npy")
    N, D = X.shape

    # Precompute squared L2 norms for each row
    sq = np.einsum("ij,ij->i", X, X)

    total_upper = 0.0
    B = 500  # Chunk size to remain well within 128 MiB memory limit

    for i in range(0, N, B):
        b = min(B, N - i)
        # Compute dot products between chunk rows and all rows from index i onwards
        G = X[i : i + b] @ X[i:].T

        # In-place distance calculation: ||x_u - x_v||^2 = ||x_u||^2 + ||x_v||^2 - 2(x_u . x_v)
        G *= -2.0
        G += sq[i : i + b, None]
        G += sq[None, i:]
        np.maximum(G, 0.0, out=G)
        np.sqrt(G, out=G)

        # Strictly upper-triangular pairs within the block (u < v)
        total_upper += float(np.triu(G[:, :b], k=1).sum(dtype=np.float64))

        # Remaining columns where global column index > global row index
        if b < G.shape[1]:
            total_upper += float(G[:, b:].sum(dtype=np.float64))

        del G

    # By symmetry d(u, v) = d(v, u), and diagonal entries d(u, u) = 0
    total = 2.0 * total_upper
    print(f"TOTAL:{total}")


if __name__ == "__main__":
    main()
