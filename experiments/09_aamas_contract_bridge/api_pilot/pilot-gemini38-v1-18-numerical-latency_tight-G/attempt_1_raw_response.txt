An efficient implementation in Python that balances memory usage and execution speed using BLAS matrix multiplication in double precision:

```python
import numpy as np


def main():
    # Load input array
    vectors = np.load("vectors.npy")
    X = vectors.astype(np.float64)
    del vectors

    # Squared L2 norms of each vector
    sq_norms = np.sum(X**2, axis=1)

    # Compute Gram matrix G = X @ X.T using BLAS dgemm
    G = X @ X.T
    del X

    # In-place calculation of squared Euclidean distances:
    # ||u - v||^2 = ||u||^2 + ||v||^2 - 2 <u, v>
    G *= -2.0
    G += sq_norms[:, None]
    G += sq_norms[None, :]

    # Ensure diagonal is exactly 0 and numerical precision errors don't cause negatives
    np.fill_diagonal(G, 0.0)
    np.maximum(G, 0.0, out=G)

    # Element-wise square root to get Euclidean distances
    np.sqrt(G, out=G)

    total = float(np.sum(G))
    print(f"TOTAL:{total}")


if __name__ == "__main__":
    main()
```
