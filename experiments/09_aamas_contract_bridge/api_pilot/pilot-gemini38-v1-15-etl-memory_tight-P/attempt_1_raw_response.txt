An implementation suited to the execution contract:

```python
import csv
import json


def main():
    totals = {}
    with open('transactions.csv', 'r', encoding='utf-8-sig', newline='') as f:
        reader = csv.reader(f)
        try:
            header = next(reader)
        except StopIteration:
            print('TOTAL:{}')
            return

        col_map = {col.strip(): idx for idx, col in enumerate(header)}
        acc_idx = col_map['account_id']
        cat_idx = col_map['category']
        amt_idx = col_map['amount_cents']

        for row in reader:
            if not row:
                continue
            acc = int(row[acc_idx])
            if acc % 11 < 7:
                cat = row[cat_idx]
                amt = int(row[amt_idx])
                val = amt * ((acc % 97) + 1)
                totals[cat] = totals.get(cat, 0) + val

    print('TOTAL:' + json.dumps(totals, sort_keys=True))


if __name__ == '__main__':
    main()
```
