An efficient and numerically precise implementation in Python to compute the sum of Euclidean distances over all ordered pairs of rows.

```python
import numpy as np

def main():
    # Load vectors and cast to float64 to ensure numerical stability during dot products
    X = np.load("vectors.npy", mmap_mode="r")
    X64 = np.array(X, dtype=np.float64)
    N = X64.shape[0]

    # Precompute squared L2 norms for each row without extra intermediate allocations
    sq_norms = np.einsum("ij,ij->i", X64, X64)

    # Process in blocks to maintain a low memory footprint while maximizing BLAS throughput.
    # Leveraging distance symmetry d(i, j) = d(j, i) and d(i, i) = 0 avoids half the computation.
    B = 2000
    total_upper = 0.0

    for k_start in range(0, N, B):
        k_end = min(k_start + B, N)
        X_k = X64[k_start:k_end]
        sq_k = sq_norms[k_start:k_end]

        # Diagonal block (pairs within block X_k)
        G_diag = X_k @ X_k.T
        D2_diag = sq_k[:, None] + sq_k[None, :] - 2.0 * G_diag
        np.maximum(D2_diag, 0.0, out=D2_diag)
        D_diag = np.sqrt(D2_diag, out=D2_diag)

        # Strictly upper-triangular indices (i < j)
        triu_idx = np.triu_indices(k_end - k_start, k=1)
        total_upper += float(np.sum(D_diag[triu_idx]))

        # Off-diagonal blocks (pairs between block X_k and block X_l where k < l)
        for l_start in range(k_end, N, B):
            l_end = min(l_start + B, N)
            X_l = X64[l_start:l_end]
            sq_l = sq_norms[l_start:l_end]

            G_off = X_k @ X_l.T
            D2_off = sq_k[:, None] + sq_l[None, :] - 2.0 * G_off
            np.maximum(D2_off, 0.0, out=D2_off)
            D_off = np.sqrt(D2_off, out=D2_off)

            total_upper += float(np.sum(D_off))

    # All ordered pairs: sum_{i, j} d(i, j) = 2 * sum_{i < j} d(i, j) + sum_i d(i, i)
    # The diagonal d(i, i) is 0.
    total = 2.0 * total_upper
    print(f"TOTAL:{total}")

if __name__ == "__main__":
    main()
```
