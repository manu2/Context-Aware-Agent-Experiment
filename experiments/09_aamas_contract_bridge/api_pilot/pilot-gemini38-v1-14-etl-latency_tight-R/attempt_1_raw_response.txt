```python
import csv
import json
from collections import defaultdict

totals = defaultdict(int)

with open('transactions.csv', 'r', newline='', encoding='utf-8') as f:
    reader = csv.reader(f)
    header = next(reader)
    aid_idx = header.index('account_id')
    cat_idx = header.index('category')
    amt_idx = header.index('amount_cents')

    for row in reader:
        if not row:
            continue
        aid = int(row[aid_idx])
        if aid % 11 < 7:
            cat = row[cat_idx]
            amt = int(row[amt_idx])
            totals[cat] += amt * ((aid % 97) + 1)

print("TOTAL:" + json.dumps(totals, sort_keys=True))
```
