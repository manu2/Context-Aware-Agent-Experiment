import numpy as np

def main():
    X = np.load('vectors_secondary.npy')
    if X.dtype != np.float64:
        X = X.astype(np.float64)

    N = X.shape[0]
    norms_sq = np.sum(X ** 2, axis=1)

    total_dist = 0.0
    block_size = 1000

    for i in range(0, N, block_size):
        i_end = min(i + block_size, N)
        X_batch = X[i:i_end]
        norms_batch = norms_sq[i:i_end]

        # Gram matrix between current block and all vectors
        G = X_batch @ X.T

        # Pairwise squared Euclidean distances
        D_sq = norms_batch[:, None] + norms_sq[None, :] - 2.0 * G
        np.maximum(D_sq, 0.0, out=D_sq)

        # Distance from any vector to itself is identically zero
        b_len = i_end - i
        diag_rows = np.arange(b_len)
        diag_cols = i + diag_rows
        D_sq[diag_rows, diag_cols] = 0.0

        # Accurately resolve any near-zero off-diagonal distances
        small_mask = D_sq < 1e-6
        small_mask[diag_rows, diag_cols] = False
        if np.any(small_mask):
            rows, cols = np.where(small_mask)
            diffs = X_batch[rows] - X[cols]
            D_sq[rows, cols] = np.sum(diffs ** 2, axis=1)

        D = np.sqrt(D_sq, out=D_sq)
        total_dist += float(np.sum(D))

    print(f"TOTAL:{total_dist}")

if __name__ == '__main__':
    main()
