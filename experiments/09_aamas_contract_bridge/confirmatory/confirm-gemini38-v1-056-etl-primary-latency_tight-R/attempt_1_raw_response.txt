import csv
import json
from collections import defaultdict


def main():
    category_sums = defaultdict(int)

    with open("transactions.csv", mode="r", encoding="utf-8", newline="") as f:
        reader = csv.reader(f)
        header = next(reader)
        acc_idx = header.index("account_id")
        cat_idx = header.index("category")
        amt_idx = header.index("amount_cents")

        for row in reader:
            if not row:
                continue
            acc_id = int(row[acc_idx])
            if acc_id % 11 < 7:
                amt = int(row[amt_idx])
                multiplier = (acc_id % 97) + 1
                category_sums[row[cat_idx]] += amt * multiplier

    output_json = json.dumps(category_sums, sort_keys=True)
    print(f"TOTAL:{output_json}")


if __name__ == "__main__":
    main()
