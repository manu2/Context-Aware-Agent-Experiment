```python
import pandas as pd
import json
from collections import defaultdict

totals = defaultdict(int)

usecols = ["account_id", "category", "amount_cents"]
dtypes = {"account_id": "int64", "amount_cents": "int64", "category": "object"}

for chunk in pd.read_csv(
    "transactions.csv",
    usecols=usecols,
    dtype=dtypes,
    chunksize=200_000,
):
    mask = (chunk["account_id"].to_numpy() % 11) < 7
    if not mask.any():
        continue
    sub = chunk.loc[mask]
    acc = sub["account_id"].to_numpy()
    amt = sub["amount_cents"].to_numpy()
    weight = (acc % 97) + 1
    weighted = amt.astype(object) * weight.astype(object)

    cats = sub["category"].to_numpy()
    grp = pd.Series(weighted, index=cats).groupby(level=0).sum()

    for cat, val in grp.items():
        totals[cat] += int(val)

result = {k: totals[k] for k in sorted(totals.keys())}
print("TOTAL:" + json.dumps(result))
```
