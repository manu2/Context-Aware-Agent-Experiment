```python
import json
import pandas as pd
import numpy as np

totals = {}

usecols = ["account_id", "category", "amount_cents"]
dtypes = {"account_id": "int64", "amount_cents": "int64", "category": "category"}

chunksize = 500_000

for chunk in pd.read_csv(
    "transactions_secondary.csv",
    usecols=usecols,
    dtype=dtypes,
    chunksize=chunksize,
):
    acc = chunk["account_id"].to_numpy()
    mask = (acc % 13) < 8
    if not mask.any():
        continue

    acc_f = acc[mask]
    amt_f = chunk["amount_cents"].to_numpy()[mask]
    cat_f = chunk["category"].to_numpy()[mask]

    mult = (acc_f % 89) + 3
    weighted = amt_f.astype(np.int64) * mult.astype(np.int64)

    s = pd.Series(weighted).groupby(cat_f, observed=True).sum()

    for cat, val in s.items():
        totals[cat] = totals.get(cat, 0) + int(val)

result = {str(k): totals[k] for k in sorted(totals.keys(), key=str)}

print("TOTAL:" + json.dumps(result))
```
