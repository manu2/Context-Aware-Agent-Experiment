import numpy as np

def main():
    # Memory-map the input to avoid loading the whole file at once unnecessarily
    X_mm = np.load('vectors_secondary.npy', mmap_mode='r')
    n, d = X_mm.shape

    # Convert to float64 for numerically stable dot products; this is only
    # 8000*1024*8 bytes ~= 64MB, which is small.
    X = np.asarray(X_mm, dtype=np.float64)

    # Precompute squared norms of each row.
    norms = np.einsum('ij,ij->i', X, X)

    total = 0.0
    block = 400  # tune to balance memory and speed; block*n*8 bytes per temp matrix

    for start in range(0, n, block):
        end = min(start + block, n)
        Xi = X[start:end]                      # (b, d)
        dot = Xi @ X.T                         # (b, n) float64
        d2 = norms[start:end, None] + norms[None, :] - 2.0 * dot
        # Numerical noise can make small negatives; clip them to zero.
        np.maximum(d2, 0.0, out=d2)
        np.sqrt(d2, out=d2)
        total += d2.sum()

        # free explicitly to help garbage collector on large temp arrays
        del dot, d2, Xi

    print(f"TOTAL:{total:.10f}")

if __name__ == '__main__':
    main()
