import numpy as np


def main():
    # Load the vectors from disk without modifying the input file
    vectors = np.load("vectors.npy")

    # Cast to float64 to maintain high numerical precision
    X = vectors.astype(np.float64)

    # Compute squared norms of each row vector: ||x_i||^2
    sq_norms = np.sum(X**2, axis=1)

    # Compute pairwise squared Euclidean distances using the identity:
    # ||x_i - x_j||^2 = ||x_i||^2 + ||x_j||^2 - 2 * <x_i, x_j>
    dists_sq = X @ X.T
    dists_sq *= -2.0
    dists_sq += sq_norms[:, None]
    dists_sq += sq_norms[None, :]

    # Set exact zeros on diagonal and clamp small negative values from floating-point noise
    np.fill_diagonal(dists_sq, 0.0)
    np.maximum(dists_sq, 0.0, out=dists_sq)

    # Compute Euclidean distances
    dists = np.sqrt(dists_sq, out=dists_sq)

    # Sum all ordered pairs
    total = np.sum(dists)

    print(f"TOTAL:{total}")


if __name__ == "__main__":
    main()
