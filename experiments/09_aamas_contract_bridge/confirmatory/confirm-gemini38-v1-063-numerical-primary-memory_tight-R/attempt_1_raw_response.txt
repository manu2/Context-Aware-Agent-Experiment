```python
import numpy as np


def main():
    vectors = np.load("vectors.npy")
    n = len(vectors)

    total_half = 0.0
    for i in range(n - 1):
        diff = vectors[i + 1 :] - vectors[i]
        dists = np.linalg.norm(diff, axis=1)
        total_half += dists.sum(dtype=np.float64)

    total = 2.0 * total_half
    print(f"TOTAL:{total}")


if __name__ == "__main__":
    main()
```
