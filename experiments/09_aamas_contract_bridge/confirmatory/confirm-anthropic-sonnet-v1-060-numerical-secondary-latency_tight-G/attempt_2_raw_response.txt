import numpy as np

def main():
    # Load data without modifying it
    data = np.load("vectors_secondary.npy", mmap_mode="r")
    n, d = data.shape

    # Work in float64 for numerically stable, exact-as-possible distance computation.
    # Convert once; 8000x1024 float64 is only ~64MB, easily fits in memory.
    X = np.array(data, dtype=np.float64)

    # Precompute squared norms of each row.
    norms = np.einsum('ij,ij->i', X, X)

    total = 0.0
    block_size = 1000  # tune for memory/time balance: block_size * n * 8 bytes per block

    Xt = X.T  # reuse transposed view for matmul

    for start in range(0, n, block_size):
        end = min(start + block_size, n)
        block = X[start:end]                     # (b, d)
        # Dot products between block rows and all rows via BLAS (fast).
        dot = block @ Xt                          # (b, n)
        # Squared Euclidean distances using ||a||^2 + ||b||^2 - 2 a.b
        sq = norms[start:end, None] + norms[None, :] - 2.0 * dot
        # Guard against tiny negative values due to floating point round-off.
        np.maximum(sq, 0.0, out=sq)
        dist = np.sqrt(sq)
        total += float(dist.sum())

    print(f"TOTAL:{total!r}")

if __name__ == "__main__":
    main()
