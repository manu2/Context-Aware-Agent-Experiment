An efficient implementation that resolves the timeout by leveraging multi-threaded BLAS matrix multiplication (`GEMM`) in 64-bit precision to compute all pairwise squared Euclidean distances simultaneously via $\|x_i - x_j\|^2 = \|x_i\|^2 + \|x_j\|^2 - 2\langle x_i, x_j \rangle$.

```python
import numpy as np


def main():
    # Load the vectors and promote to float64 for full numerical precision
    vectors = np.load("vectors.npy")
    X = vectors.astype(np.float64)

    # Compute squared L2 norms for each row
    sq_norms = np.sum(X * X, axis=1)

    # Compute Gram matrix using high-performance GEMM: G = X @ X.T
    G = X @ X.T

    # Form squared distance matrix in-place: D^2 = ||x_i||^2 + ||x_j||^2 - 2 * (x_i . x_j)
    G *= -2.0
    G += sq_norms[:, None]
    G += sq_norms[None, :]

    # Zero the diagonal exactly
    np.fill_diagonal(G, 0.0)

    # Clamp any small negative values due to floating-point roundoff to 0
    np.maximum(G, 0.0, out=G)

    # Compute Euclidean distance matrix in-place
    np.sqrt(G, out=G)

    total = float(G.sum())
    print(f"TOTAL:{total}")


if __name__ == "__main__":
    main()
```
