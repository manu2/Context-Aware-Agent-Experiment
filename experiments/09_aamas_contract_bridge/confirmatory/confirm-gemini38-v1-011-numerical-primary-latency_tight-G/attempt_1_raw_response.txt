import numpy as np


def main():
    X = np.load("vectors.npy").astype(np.float64)

    # Compute squared norms: shape (8000, 1)
    sq_norms = np.sum(X**2, axis=1, keepdims=True)

    # Compute Gram matrix G = X @ X.T
    G = X @ X.T

    # Compute squared Euclidean distance matrix in-place: D^2 = ||x_i||^2 + ||x_j||^2 - 2 <x_i, x_j>
    G *= -2.0
    G += sq_norms
    G += sq_norms.T

    # Ensure zero diagonal and handle small precision issues
    np.fill_diagonal(G, 0.0)
    np.maximum(G, 0.0, out=G)

    # Compute Euclidean distances in-place
    np.sqrt(G, out=G)

    total = float(np.sum(G))
    print(f"TOTAL:{total}")


if __name__ == "__main__":
    main()
