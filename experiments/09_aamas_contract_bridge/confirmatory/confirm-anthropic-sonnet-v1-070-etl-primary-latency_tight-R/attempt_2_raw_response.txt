```python
import pandas as pd
import numpy as np
import json

def main():
    df = pd.read_csv(
        "transactions.csv",
        usecols=["account_id", "category", "amount_cents"],
        dtype={"account_id": "int64", "amount_cents": "int64", "category": "string"},
    )

    account_id = df["account_id"].to_numpy(dtype=np.int64)
    amount = df["amount_cents"].to_numpy(dtype=np.int64)
    category = df["category"].to_numpy()

    mask = (account_id % 11) < 7
    if not mask.any():
        print("TOTAL:" + json.dumps({}))
        return

    account_id = account_id[mask]
    amount = amount[mask]
    category = category[mask]

    multiplier = (account_id % 97) + 1
    contrib = amount.astype(object) * multiplier.astype(object)

    result = pd.Series(contrib, index=category).groupby(level=0).sum()

    out = {str(k): int(v) for k, v in result.items()}
    out_sorted = dict(sorted(out.items()))

    print("TOTAL:" + json.dumps(out_sorted))

if __name__ == "__main__":
    main()
```
