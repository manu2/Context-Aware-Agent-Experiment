An efficient and exact Python program to compute the total sum of Euclidean distances:

```python
import numpy as np


def main():
    vectors = np.load("vectors_secondary.npy").astype(np.float64)
    n = len(vectors)
    total = 0.0

    for i in range(n - 1):
        diff = vectors[i + 1 :] - vectors[i]
        diff *= diff
        s = diff.sum(axis=1)
        np.sqrt(s, out=s)
        total += s.sum()

    total *= 2.0
    print(f"TOTAL:{total}")


if __name__ == "__main__":
    main()
```
