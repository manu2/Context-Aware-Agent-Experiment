import numpy as np

def main():
    vectors = np.load('vectors.npy')  # float32, shape (8000, 1024)
    n = vectors.shape[0]

    block_size = 400

    # Compute norms (sum of squares per row) in float64 for accuracy,
    # processed in blocks to avoid large temporary allocations.
    norms = np.empty(n, dtype=np.float64)
    for start in range(0, n, block_size):
        end = min(start + block_size, n)
        blk64 = vectors[start:end].astype(np.float64)
        norms[start:end] = np.einsum('ij,ij->i', blk64, blk64)
        del blk64

    total = 0.0

    vt = vectors.T  # view, no copy

    for start in range(0, n, block_size):
        end = min(start + block_size, n)
        block = vectors[start:end]  # float32, (b, 1024)

        # float32 matmul (fast BLAS sgemm), then upcast result to float64
        dots = block @ vt  # float32, shape (b, n)
        dots64 = dots.astype(np.float64)
        del dots

        sq = norms[start:end][:, None] + norms[None, :] - 2.0 * dots64
        del dots64

        np.maximum(sq, 0.0, out=sq)
        np.sqrt(sq, out=sq)

        total += sq.sum()
        del sq

    print(f"TOTAL:{total}")

if __name__ == "__main__":
    main()
