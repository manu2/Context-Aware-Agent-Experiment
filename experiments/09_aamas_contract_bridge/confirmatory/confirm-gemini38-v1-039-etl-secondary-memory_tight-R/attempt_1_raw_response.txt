```python
import csv
from collections import defaultdict
import json


def main():
    totals = defaultdict(int)
    with open("transactions_secondary.csv", "r", encoding="utf-8", newline="") as f:
        reader = csv.reader(f)
        header = next(reader)
        col_map = {col.strip(): i for i, col in enumerate(header)}
        acc_idx = col_map["account_id"]
        cat_idx = col_map["category"]
        amt_idx = col_map["amount_cents"]

        for row in reader:
            if not row:
                continue
            acc_id = int(row[acc_idx])
            if acc_id % 13 < 8:
                cat = row[cat_idx]
                amt = int(row[amt_idx])
                totals[cat] += amt * ((acc_id % 89) + 3)

    result_json = json.dumps(totals, sort_keys=True)
    print(f"TOTAL:{result_json}")


if __name__ == "__main__":
    main()
```
