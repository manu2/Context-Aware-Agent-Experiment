import numpy as np

def main():
    X = np.load('vectors.npy', mmap_mode='r')  # (8000, 1024) float32
    n = X.shape[0]

    # Load into memory as float32 (8000*1024*4 bytes ~ 32MB, fine)
    X = np.asarray(X, dtype=np.float32)

    # Compute squared norms in float64 for accuracy
    # sum of squares per row
    sq_norms = np.einsum('ij,ij->i', X, X, dtype=np.float64)

    total = np.float64(0.0)

    block_size = 500  # tune for memory/time balance
    XT = X.T  # (1024, 8000), view, no copy

    for start in range(0, n, block_size):
        end = min(start + block_size, n)
        block = X[start:end]  # (b, 1024) float32

        # Compute dot products block @ X.T -> (b, n) using BLAS (float32 for speed)
        dots = block.astype(np.float32) @ XT  # (b, n) float32
        dots = dots.astype(np.float64)

        # squared distances: ||a||^2 + ||b||^2 - 2*a.b
        a_norms = sq_norms[start:end].reshape(-1, 1)  # (b,1)
        b_norms = sq_norms.reshape(1, -1)  # (1,n)

        sq_dist = a_norms + b_norms - 2.0 * dots
        # numerical safety: clip small negative values to zero
        np.maximum(sq_dist, 0.0, out=sq_dist)

        dist = np.sqrt(sq_dist)

        total += dist.sum(dtype=np.float64)

    # total is a Python float via numpy float64; format as decimal
    print(f"TOTAL:{total:.10f}")

if __name__ == '__main__':
    main()
