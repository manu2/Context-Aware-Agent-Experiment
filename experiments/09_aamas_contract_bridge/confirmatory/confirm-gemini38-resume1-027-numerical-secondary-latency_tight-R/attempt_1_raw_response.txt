An efficient and complete Python program to compute the exact sum of Euclidean distances across all ordered pairs of rows:

```python
import numpy as np


def main():
    # Load the array and convert to float64 for numerical precision
    vectors = np.load("vectors_secondary.npy").astype(np.float64)

    # Compute Gram matrix (all pairwise dot products)
    dots = vectors @ vectors.T

    # Extract squared norms before in-place modifications
    sq_norms = np.diag(dots).copy()

    # Compute squared Euclidean distances in-place: ||u - v||^2 = ||u||^2 + ||v||^2 - 2*(u . v)
    dots *= -2.0
    dots += sq_norms[:, None]
    dots += sq_norms[None, :]

    # Handle numerical roundoff at or near zero
    np.maximum(dots, 0.0, out=dots)
    np.fill_diagonal(dots, 0.0)

    # Compute Euclidean distances
    np.sqrt(dots, out=dots)

    # Sum distances over all ordered pairs
    total = float(np.sum(dots))

    print(f"TOTAL:{total}")


if __name__ == "__main__":
    main()
```
