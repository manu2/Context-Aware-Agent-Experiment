An efficient implementation in Python that balances memory usage and execution speed using NumPy and block-wise matrix operations:

```python
import numpy as np


def main():
    # Load the vectors and cast to float64 to ensure high numerical precision
    X = np.load("vectors_secondary.npy").astype(np.float64)
    n = X.shape[0]
    sq_norm = np.sum(X**2, axis=1)

    total_upper = 0.0
    block_size = 2000

    # Exploit symmetry: D_ij = D_ji and D_ii = 0.
    # Total sum over all ordered pairs = 2 * sum_{i < j} D_ij.
    for i in range(0, n, block_size):
        X_i = X[i : i + block_size]
        sq_i = sq_norm[i : i + block_size, None]

        for j in range(i, n, block_size):
            X_j = X[j : j + block_size]
            sq_j = sq_norm[j : j + block_size]

            # Compute squared Euclidean distance using ||a - b||^2 = ||a||^2 + ||b||^2 - 2<a, b>
            gram = X_i @ X_j.T
            d2 = sq_i + sq_j[None, :] - 2.0 * gram
            np.maximum(d2, 0.0, out=d2)
            d = np.sqrt(d2)

            if i == j:
                total_upper += np.sum(np.triu(d, k=1))
            else:
                total_upper += np.sum(d)

    total = 2.0 * total_upper
    print(f"TOTAL:{total}")


if __name__ == "__main__":
    main()
```
