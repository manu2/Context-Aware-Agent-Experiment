import pandas as pd
import json
from collections import defaultdict

totals = defaultdict(int)

for chunk in pd.read_csv(
    "transactions.csv",
    usecols=["account_id", "category", "amount_cents"],
    dtype={"account_id": "int64", "amount_cents": "int64", "category": "str"},
    chunksize=200_000,
):
    mask = (chunk["account_id"] % 11) < 7
    if not mask.any():
        continue
    sub = chunk.loc[mask]
    multiplier = (sub["account_id"] % 97) + 1
    weighted = sub["amount_cents"] * multiplier
    grouped = weighted.groupby(sub["category"]).sum()
    for cat, val in grouped.items():
        totals[cat] += int(val)

result = {cat: totals[cat] for cat in sorted(totals.keys())}
print("TOTAL:" + json.dumps(result))
