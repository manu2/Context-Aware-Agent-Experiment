import numpy as np


def main():
    # Load vectors
    vectors = np.load("vectors.npy")

    # Cast to float64 for numerical precision and prevent overflow
    vectors_64 = vectors.astype(np.float64)

    # Compute squared L2 norms of each vector
    sq_norms = np.sum(vectors_64**2, axis=1)

    # Compute Gram matrix G = X @ X.T using BLAS
    # Distances squared: ||x_i - x_j||^2 = ||x_i||^2 + ||x_j||^2 - 2 <x_i, x_j>
    dists_sq = vectors_64 @ vectors_64.T
    dists_sq *= -2.0
    dists_sq += sq_norms[:, None]
    dists_sq += sq_norms[None, :]

    # Numerical stability: clamp small negative values to 0.0 and zero out diagonal
    np.maximum(dists_sq, 0.0, out=dists_sq)
    np.fill_diagonal(dists_sq, 0.0)

    # Compute Euclidean distances and their sum
    np.sqrt(dists_sq, out=dists_sq)
    total_dist = float(dists_sq.sum())

    print(f"TOTAL:{total_dist}")


if __name__ == "__main__":
    main()
