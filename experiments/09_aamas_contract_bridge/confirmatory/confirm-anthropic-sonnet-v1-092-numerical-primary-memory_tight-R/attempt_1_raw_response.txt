```python
import numpy as np

def main():
    X = np.load('vectors.npy').astype(np.float64)
    n = X.shape[0]

    block_size = 200
    total = 0.0

    for start in range(0, n, block_size):
        end = min(start + block_size, n)
        block = X[start:end]  # shape (b, d)

        # compute squared distances between block and all rows
        # ||a-b||^2 = ||a||^2 + ||b||^2 - 2 a.b
        diff_sq = (block[:, None, :] - X[None, :, :]) ** 2
        dist = np.sqrt(diff_sq.sum(axis=2))
        total += dist.sum()

    # total is a numpy float64 scalar; convert to Decimal-like exact printing
    print(f"TOTAL:{total:.10f}")

if __name__ == "__main__":
    main()
```
