```python
import numpy as np


def main():
    # Load input array and convert to float64 to ensure high numerical precision
    # during dot product and norm expansions.
    X = np.load("vectors_secondary.npy").astype(np.float64)
    n = X.shape[0]

    # Precompute squared L2 norms for each vector: ||x_i||^2
    sq_norms = np.sum(X**2, axis=1)

    total_distance = 0.0
    chunk_size = 1000

    # Process in chunks to maintain a low memory footprint while utilizing BLAS GEMM
    for start in range(0, n, chunk_size):
        end = min(start + chunk_size, n)
        chunk_len = end - start

        # Compute dot products between chunk vectors and all vectors: X_chunk @ X^T
        G = X[start:end] @ X.T

        # ||u - v||^2 = ||u||^2 + ||v||^2 - 2 * <u, v>
        d_sq = sq_norms[start:end, None] + sq_norms[None, :] - 2.0 * G

        # Set diagonal entries strictly to zero to avoid precision noise where i == j
        d_sq[np.arange(chunk_len), np.arange(start, end)] = 0.0

        # Guard against small negative values from floating-point roundoff
        np.maximum(d_sq, 0.0, out=d_sq)
        np.sqrt(d_sq, out=d_sq)

        total_distance += np.sum(d_sq)

    print(f"TOTAL:{total_distance}")


if __name__ == "__main__":
    main()
```
