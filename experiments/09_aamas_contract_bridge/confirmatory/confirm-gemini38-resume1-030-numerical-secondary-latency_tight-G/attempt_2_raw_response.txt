import numpy as np


def main():
    # Load the vectors
    vectors = np.load("vectors_secondary.npy")

    # Cast to float64 to ensure high precision and prevent numerical cancellation
    vectors_64 = vectors.astype(np.float64)
    sq_norms = np.sum(vectors_64**2, axis=1)

    n = vectors_64.shape[0]
    total = 0.0

    # Process in chunks to maintain low memory footprint while leveraging BLAS GEMM
    chunk_size = 2000
    for start in range(0, n, chunk_size):
        end = min(start + chunk_size, n)
        chunk = vectors_64[start:end]

        # Compute squared Euclidean distances: ||u - v||^2 = ||u||^2 + ||v||^2 - 2 <u, v>
        dists_sq = chunk @ vectors_64.T
        dists_sq *= -2.0
        dists_sq += sq_norms[start:end, None]
        dists_sq += sq_norms[None, :]

        # Clamp small negative values caused by floating-point roundoff
        np.maximum(dists_sq, 0.0, out=dists_sq)

        # Ensure diagonal elements (distance to self) are strictly zero
        k = np.arange(end - start)
        dists_sq[k, start + k] = 0.0

        dists = np.sqrt(dists_sq, out=dists_sq)
        total += float(dists.sum(dtype=np.float64))

    print(f"TOTAL:{total}")


if __name__ == "__main__":
    main()
