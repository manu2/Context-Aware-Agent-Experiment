import numpy as np

def main():
    X = np.load('vectors.npy', mmap_mode='r')  # (8000, 1024) float32
    n = X.shape[0]

    # Convert once to float64 for accurate distance computation
    Xf = np.asarray(X, dtype=np.float64)

    # Precompute squared norms
    norms = np.einsum('ij,ij->i', Xf, Xf)

    total = np.float64(0.0)
    block_size = 500

    for start in range(0, n, block_size):
        end = min(start + block_size, n)
        block = Xf[start:end]                     # (b, 1024)
        dot = block @ Xf.T                         # (b, n) float64, BLAS dgemm
        dist_sq = norms[start:end, None] + norms[None, :] - 2.0 * dot
        np.maximum(dist_sq, 0.0, out=dist_sq)      # guard against tiny negatives
        dist = np.sqrt(dist_sq, out=dist_sq)
        total += dist.sum(dtype=np.float64)

    # Print with full precision decimal representation
    print(f"TOTAL:{total:.10f}")

if __name__ == '__main__':
    main()
