import numpy as np

def main():
    X = np.load("vectors_secondary.npy")
    X = X.astype(np.float64, copy=False)
    n = X.shape[0]

    norms = np.sum(X * X, axis=1)

    total = 0.0
    chunk_size = 500

    for start in range(0, n, chunk_size):
        end = min(start + chunk_size, n)
        Xc = X[start:end]
        # dot products between chunk rows and all rows
        G = Xc @ X.T  # shape (chunk, n)
        d2 = norms[start:end][:, None] + norms[None, :] - 2.0 * G
        # numerical safety: clip small negative values to zero
        np.maximum(d2, 0.0, out=d2)
        d = np.sqrt(d2)
        total += np.sum(d, dtype=np.float64)

    # total is a numpy float64 scalar; convert to Python float for printing
    total_val = float(total)
    print(f"TOTAL:{total_val:.10f}")

if __name__ == "__main__":
    main()
