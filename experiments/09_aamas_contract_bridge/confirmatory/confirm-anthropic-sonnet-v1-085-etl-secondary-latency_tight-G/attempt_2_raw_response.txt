```python
import pandas as pd
import json
from collections import defaultdict

totals = defaultdict(int)

cols = ['account_id', 'category', 'amount_cents']
dtypes = {'account_id': 'int64', 'amount_cents': 'int64', 'category': 'category'}

chunksize = 500_000

for chunk in pd.read_csv(
    'transactions_secondary.csv',
    usecols=cols,
    dtype=dtypes,
    chunksize=chunksize,
    engine='c',
    memory_map=True,
):
    mask = (chunk['account_id'].values % 13) < 8
    if not mask.any():
        continue
    sub = chunk.loc[mask]
    weights = (sub['account_id'].values % 89) + 3
    contrib = sub['amount_cents'].values.astype('int64') * weights.astype('int64')

    grouped = pd.Series(contrib, index=sub['category'].values).groupby(level=0).sum()

    for cat, val in grouped.items():
        totals[cat] += int(val)

result = {k: totals[k] for k in sorted(totals.keys())}
print("TOTAL:" + json.dumps(result))
```
