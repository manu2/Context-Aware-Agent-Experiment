```python
import csv
import json

totals = {}

with open('transactions.csv', mode='r', encoding='utf-8', newline='') as f:
    reader = csv.reader(f)
    header = next(reader)
    acc_idx = header.index('account_id')
    cat_idx = header.index('category')
    amt_idx = header.index('amount_cents')

    for row in reader:
        if not row:
            continue
        acc_id = int(row[acc_idx])
        if acc_id % 11 < 7:
            category = row[cat_idx]
            amt = int(row[amt_idx])
            weight = (acc_id % 97) + 1
            totals[category] = totals.get(category, 0) + amt * weight

sorted_totals = dict(sorted(totals.items()))
print(f"TOTAL:{json.dumps(sorted_totals, sort_keys=True)}")
```
