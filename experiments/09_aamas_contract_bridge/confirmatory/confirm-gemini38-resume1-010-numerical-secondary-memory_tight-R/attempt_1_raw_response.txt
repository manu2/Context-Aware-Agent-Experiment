An efficient and exact Python program to compute the sum of Euclidean distances over all ordered pairs:

```python
import numpy as np


def main():
    # Load vectors from the current directory
    vectors = np.load("vectors_secondary.npy")

    # Cast to float64 to ensure high numerical precision
    X = vectors.astype(np.float64)

    # Compute Gram matrix (pairwise dot products)
    G = X @ X.T

    # Squared Euclidean distance: ||u - v||^2 = ||u||^2 + ||v||^2 - 2*(u . v)
    sq_norms = np.diag(G)
    D2 = sq_norms[:, None] + sq_norms[None, :] - 2.0 * G

    # Numerical safeguard for tiny negative values due to floating-point rounding
    np.maximum(D2, 0.0, out=D2)

    # Compute Euclidean distances
    D = np.sqrt(D2)

    # Diagonal entries are exactly zero
    np.fill_diagonal(D, 0.0)

    total = float(np.sum(D))
    print(f"TOTAL:{total}")


if __name__ == "__main__":
    main()
```
