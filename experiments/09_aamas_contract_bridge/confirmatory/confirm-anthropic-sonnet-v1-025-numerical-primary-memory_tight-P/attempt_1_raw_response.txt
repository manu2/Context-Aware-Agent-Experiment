import numpy as np

def main():
    vectors = np.load("vectors.npy", mmap_mode='r')  # avoid full copy in memory upfront
    n, d = vectors.shape

    # Compute squared norms in float64 for accuracy
    norms = np.empty(n, dtype=np.float64)
    block_norm = 500
    for start in range(0, n, block_norm):
        end = min(start + block_norm, n)
        chunk = np.asarray(vectors[start:end], dtype=np.float64)
        norms[start:end] = np.einsum('ij,ij->i', chunk, chunk)

    total = 0.0
    B = 100  # chunk size for row blocks

    # Keep a float32 copy of full vectors for matmul (already on disk, but need in-memory for dot products)
    # Load fully once as float32 (32MB), acceptable within memory budget.
    full = np.asarray(vectors, dtype=np.float32)

    for start in range(0, n, B):
        end = min(start + B, n)
        A = full[start:end]  # (b, d) float32
        # dot products (b, n) in float32, then upcast for subtraction
        dots = A @ full.T  # float32 result
        dots = dots.astype(np.float64)

        dist2 = norms[start:end][:, None] + norms[None, :] - 2.0 * dots
        np.maximum(dist2, 0.0, out=dist2)
        dist = np.sqrt(dist2)
        total += dist.sum()

    # Print total with fixed decimal notation, no scientific notation
    print(f"TOTAL:{total:.6f}")

if __name__ == "__main__":
    main()
