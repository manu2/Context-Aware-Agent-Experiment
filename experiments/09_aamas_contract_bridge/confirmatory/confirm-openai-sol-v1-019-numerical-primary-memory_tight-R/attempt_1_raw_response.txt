import numpy as np

vectors = np.load("vectors.npy", mmap_mode="r")
x = np.asarray(vectors, dtype=np.float64)

n = x.shape[0]
block_size = 1024
squared_norms = np.einsum("ij,ij->i", x, x, dtype=np.float64)
total = np.longdouble(0)

for i in range(0, n, block_size):
    i_end = min(i + block_size, n)
    xi = x[i:i_end]
    ni = squared_norms[i:i_end]

    for j in range(i, n, block_size):
        j_end = min(j + block_size, n)
        xj = x[j:j_end]
        nj = squared_norms[j:j_end]

        distances = xi @ xj.T
        distances *= -2.0
        distances += ni[:, None]
        distances += nj[None, :]
        np.maximum(distances, 0.0, out=distances)

        if i == j:
            np.fill_diagonal(distances, 0.0)

        np.sqrt(distances, out=distances)
        block_sum = np.sum(distances, dtype=np.longdouble)

        total += block_sum if i == j else 2 * block_sum

print("TOTAL:" + np.format_float_positional(total, unique=True, trim="-"))
