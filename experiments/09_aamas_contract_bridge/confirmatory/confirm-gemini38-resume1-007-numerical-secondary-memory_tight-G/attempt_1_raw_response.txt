import numpy as np


def main():
    # Load vectors and cast to float64 for precision during distance computation
    X = np.load("vectors_secondary.npy").astype(np.float64)
    n = X.shape[0]

    sq_norms = np.sum(X**2, axis=1)

    total = 0.0
    batch_size = 1000

    for start in range(0, n, batch_size):
        end = min(start + batch_size, n)
        X_batch = X[start:end]
        norms_batch = sq_norms[start:end]

        # Intra-batch pairs (diagonal block)
        G_diag = X_batch @ X_batch.T
        D2_diag = norms_batch[:, None] + norms_batch[None, :] - 2.0 * G_diag
        np.maximum(D2_diag, 0.0, out=D2_diag)
        np.fill_diagonal(D2_diag, 0.0)
        total += float(np.sum(np.sqrt(D2_diag)))

        # Inter-batch pairs with remaining rows (off-diagonal blocks)
        if end < n:
            X_rest = X[end:]
            norms_rest = sq_norms[end:]
            G_rest = X_batch @ X_rest.T
            D2_rest = norms_batch[:, None] + norms_rest[None, :] - 2.0 * G_rest
            np.maximum(D2_rest, 0.0, out=D2_rest)
            # Factor of 2 accounts for both ordered pairs (i, j) and (j, i)
            total += 2.0 * float(np.sum(np.sqrt(D2_rest)))

    print(f"TOTAL:{total}")


if __name__ == "__main__":
    main()
