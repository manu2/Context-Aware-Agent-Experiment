An efficient, self-contained Python script to compute the total sum of all pairwise Euclidean distances within the specified RAM and time limits is provided below.

```python
import numpy as np


def main():
    # Memory-map the input matrix to minimize memory usage
    vectors = np.load('vectors.npy', mmap_mode='r')
    n, d = vectors.shape

    # Compute squared L2 norm for each row in chunks to save memory
    block_size = 1000
    sq_norms = np.empty(n, dtype=np.float32)
    for i in range(0, n, block_size):
        i_end = min(i + block_size, n)
        sq_norms[i:i_end] = np.sum(vectors[i:i_end] ** 2, axis=1)

    total_dist = 0.0

    # Exploit symmetry: compute diagonal blocks and upper-triangular blocks
    for i in range(0, n, block_size):
        i_end = min(i + block_size, n)
        v_i = vectors[i:i_end]
        sq_i = sq_norms[i:i_end, None]

        # 1. Diagonal block (i == j)
        gram = np.dot(v_i, v_i.T)
        dist_sq = sq_i + sq_i.T - 2.0 * gram
        np.maximum(dist_sq, 0.0, out=dist_sq)
        np.fill_diagonal(dist_sq, 0.0)
        dist = np.sqrt(dist_sq, out=dist_sq)
        total_dist += np.sum(dist, dtype=np.float64)

        # 2. Off-diagonal blocks (j > i)
        for j in range(i + block_size, n, block_size):
            j_end = min(j + block_size, n)
            v_j = vectors[j:j_end]
            sq_j = sq_norms[None, j:j_end]

            gram = np.dot(v_i, v_j.T)
            dist_sq = sq_i + sq_j - 2.0 * gram
            np.maximum(dist_sq, 0.0, out=dist_sq)
            dist = np.sqrt(dist_sq, out=dist_sq)
            # Factor of 2 accounts for symmetric lower triangle (j, i)
            total_dist += 2.0 * np.sum(dist, dtype=np.float64)

    print(f"TOTAL_DIST:{total_dist}")


if __name__ == '__main__':
    main()
```
